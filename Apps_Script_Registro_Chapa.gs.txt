function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);

    if (body.token !== TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ok: false, erro: "token invalido"}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (body.tipo === "registro_chapa_consep_primavera") {
      gravarRegistroChapa(body.registro);
    } else if (body.tipo === "filiacao_consep_primavera") {
      gravarFiliacao(body.registro);
    } else if (body.tipo === "cadastro_assembleia_consep_primavera") {
      gravarCadastroAssembleia(body.registro);
    }

    return ContentService.createTextOutput(JSON.stringify({ok: true}))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ok: false, erro: String(err)}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function salvarImagemNoDrive(base64Img, nomeArquivo, nomePasta) {
  if (!base64Img || base64Img.indexOf("base64,") === -1) return "";
  try {
    var base64 = base64Img.split("base64,")[1];
    var bytes = Utilities.base64Decode(base64);
    var blob = Utilities.newBlob(bytes, "image/jpeg", nomeArquivo);
    var pastas = DriveApp.getFoldersByName(nomePasta);
    var pasta = pastas.hasNext() ? pastas.next() : DriveApp.createFolder(nomePasta);
    var arquivo = pasta.createFile(blob);
    arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return arquivo.getUrl();
  } catch (errImg) {
    return "[erro ao salvar imagem: " + errImg + "]";
  }
}

function gravarRegistroChapa(r) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(ABA_CHAPA);
  if (!sheet) {
    sheet = ss.insertSheet(ABA_CHAPA);
    sheet.appendRow(CABECALHO_CHAPA);
    sheet.setFrozenRows(1);
  }

  var linkAssinatura = salvarImagemNoDrive(
    r.assinatura, "assinatura_" + r.protocolo + ".jpg",
    "Assinaturas - Registro Chapa CONSEP Primavera"
  );

  var linkComprovante = salvarImagemNoDrive(
    r.comprovante_residencia, "comprovante_" + r.protocolo + ".jpg",
    "Comprovantes de Residencia - Registro Chapa CONSEP Primavera"
  );

  sheet.appendRow([
    r.protocolo || "",
    r.data_registro || new Date().toISOString(),
    r.cargo || "",
    r.nome || "",
    r.cpf || "",
    r.rg || "",
    r.telefone || "",
    r.vinculo_bairro || "",
    r.declaracao_maioridade ? "SIM" : "NAO",
    r.declaracao_sem_condenacao ? "SIM" : "NAO",
    r.declaracao_lgpd ? "SIM" : "NAO",
    r.declaracao_sem_outra_chapa ? "SIM" : "NAO",
    r.declaracao_sem_impedimento ? "SIM" : "NAO",
    linkComprovante,
    linkAssinatura,
    r.status || "pendente"
  ]);
}
